%=============== Typy slajdów ===============
% Pomocnicze: blok z czerwonym nagłówkiem albo same punkty, gdy nagłówek pusty
% \punktyblok{czerwony tekst}{\item ... \item ...}
\newcommand{\punktyblok}[2]{%
  \if\relax\detokenize{#1}\relax
    \begin{itemize}#2\end{itemize}%
  \else
    \begin{block}{#1}\begin{itemize}#2\end{itemize}\end{block}%
  \fi
}

% Sekcja - jedno duże słowo na środku
% \sekcja{nazwa}
\newcommand{\sekcja}[1]{%
  \section{#1}
  \begin{frame}
    \vfill
    \begin{center}
      {\fontsize{36}{42}\selectfont\color{PWRred}\textbf{#1}}
    \end{center}
    \vfill
  \end{frame}
}

% Zdjęcie - duże zdjęcie z (opcjonalnym) podpisem
% \zdjecie[opcje includegraphics]{tytuł}{plik}{podpis}
\newcommand{\zdjecie}[4][]{%
  \begin{frame}{#2}
    \begin{figure}
      \centering
      \includegraphics[#1,width=\linewidth,height=0.75\textheight,keepaspectratio]{#3}
      \if\relax\detokenize{#4}\relax\else\caption{#4}\fi
    \end{figure}
  \end{frame}
}

% Tekst/zdjęcie LP - po lewej tekst, po prawej zdjęcie
% \tekstLP[opcje includegraphics]{tytuł}{czerwony tekst}{\item ...}{plik}
\newcommand{\tekstLP}[5][]{%
  \begin{frame}{#2}
    \begin{columns}[c]
      \begin{column}{0.5\textwidth}
        \punktyblok{#3}{#4}
      \end{column}
      \begin{column}{0.45\textwidth}
        \centering
        \includegraphics[#1,width=\linewidth,height=0.7\textheight,keepaspectratio]{#5}
      \end{column}
    \end{columns}
  \end{frame}
}

% Tekst/zdjęcie GD - na górze tekst, na dole zdjęcie
% \tekstGD{tytuł}{czerwony tekst}{\item ...}{plik}
\newcommand{\tekstGD}[4]{%
  \begin{frame}{#1}
    \punktyblok{#2}{#3}
    \vfill
    \begin{center}
      \includegraphics[width=\linewidth,height=0.5\textheight,keepaspectratio]{#4}
    \end{center}
    \vfill
  \end{frame}
}
